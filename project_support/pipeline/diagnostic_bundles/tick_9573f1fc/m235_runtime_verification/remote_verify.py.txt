def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def regular(path):
    a = os.lstat(path)
    assert stat.S_ISREG(a.st_mode) and a.st_uid == 1001 and a.st_nlink == 1
    assert Path(path).resolve() == Path(path)
    return a

def run(argv, env=None):
    process = subprocess.Popen(argv, stdout=subprocess.PIPE, stderr=subprocess.PIPE, env=env)
    try:
        stdout, stderr = process.communicate(timeout=25)
    except subprocess.TimeoutExpired:
        raise RuntimeError('Read-only child timeout; preserve process, no signal or retry')
    assert process.returncode == 0, stderr.decode('utf-8','replace')[-1500:]
    return stdout

def git(*args):
    return run(['/usr/bin/git','--no-optional-locks','-c','core.fsmonitor=false','-C',str(root),*args],git_env)

def snapshot():
    head = git('rev-parse','--verify','HEAD').decode().strip()
    assert head == source
    hashes = {}
    for relative, expected in baseline['source_files_exact9'].items():
        regular(root/relative)
        original = git('cat-file','blob',source+':'+relative)
        assert hashlib.sha256(original).hexdigest() == expected
        if relative.endswith('.py'):
            hashes[relative] = sha(root/relative)
            assert hashes[relative] == expected, relative
    return head, hashes
