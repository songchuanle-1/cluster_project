def pin():
    assert socket.gethostname() == "dl04-MS-7E06" and os.getuid() == os.geteuid() == 1001
    assert datetime.now(timezone.utc) < datetime.fromisoformat(OWNER["deadline_at"])

def hash_file(path):
    info = os.lstat(path)
    assert stat.S_ISREG(info.st_mode) and info.st_uid == 1001 and info.st_size <= 256 * 1024 * 1024
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for value in iter(lambda: stream.read(65536), b""):
            pin()
            digest.update(value)
    after = os.lstat(path)
    assert (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns) == (
        after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns)
    return digest.hexdigest()


def identity(pid):
    try:
        return (Path("/proc") / str(pid) / "stat").read_text().rsplit(")", 1)[1].split()[19]
    except FileNotFoundError:
        return None


def tail(fd, maximum):
    size = os.fstat(fd).st_size
    return os.pread(fd, min(size, maximum), max(0, size - maximum)).decode("utf-8", "replace")


def observe(argv, environment, payload, seconds, global_stop, label, pass_fds=(), capture_pair=None):
    pin()
    if capture_pair is None:
        out = os.memfd_create("constructor_supervisor_stdout", os.MFD_CLOEXEC)
        err = os.memfd_create("constructor_supervisor_stderr", os.MFD_CLOEXEC)
    else:
        out, err = capture_pair
    process = subprocess.Popen(argv, cwd=str(ROOT), env=environment,
                               stdin=subprocess.PIPE, stdout=out, stderr=err, pass_fds=pass_fds)
    PROCESSES.append(process)
    start = identity(process.pid)
    if payload is not None:
        encoded = json.dumps(payload).encode()
        assert len(encoded) <= 16384
        process.stdin.write(encoded)
    process.stdin.close()
    until = min(global_stop, time.monotonic() + seconds)
    while process.poll() is None and time.monotonic() < until:
        time.sleep(0.02)
    returncode = process.poll()
    stdout = tail(out, 65536)
    records = []
    for line in stdout.splitlines():
        try:
            records.append(json.loads(line))
        except ValueError:
            pass
    result = {"label": label, "pid": process.pid, "start_ticks": start,
              "certainty": "terminal" if returncode is not None else "uncertain",
              "returncode": returncode, "stdout_bytes": os.fstat(out).st_size,
              "stderr_bytes": os.fstat(err).st_size,
              "records": records[-32:], "faulthandler_stderr_tail": tail(err, 3072),
              "stderr_truncated": os.fstat(err).st_size > 3072, "signals_sent": False}
    if returncode is None:
        result["reason"] = "SOFT_TIMEOUT_PRESERVE_PID_NO_RETRY"
        result["output_FDs_if_process_remains"] = [f"/proc/{process.pid}/fd/1", f"/proc/{process.pid}/fd/2"]
    else:
        os.close(out)
        os.close(err)
    return result, stdout


def small(path):
    pin()
    try:
        with Path(path).open("r") as stream:
            value = stream.read(8193)
        assert len(value) <= 8192
        return value.strip()
    except FileNotFoundError:
        return None


def host_snapshot():
    meminfo = small("/proc/meminfo")
    available = next(int(row.split()[1]) * 1024 for row in meminfo.splitlines()
                     if row.startswith("MemAvailable:"))
    membership = small("/proc/self/cgroup")
    group = next((row.split(":", 2)[2] for row in membership.splitlines() if row.startswith("0::")), None)
    result = {"mem_available_bytes": available, "loadavg": small("/proc/loadavg"),
              "cpu_affinity": sorted(os.sched_getaffinity(0)), "cgroup_membership": membership,
              "constructor_minimum_headroom_bytes": 4 * 1024 ** 3,
              "four_worker_budget_PASS": False, "GPU_resource_admission": False}
    shm = os.statvfs("/dev/shm")
    result["dev_shm_available_bytes"] = shm.f_bavail * shm.f_frsize
    result["cgroup_ancestors"] = []
    adequate = available >= 4 * 1024 ** 3 and bool(result["cpu_affinity"])
    if group is None:
        result["capacity_ambiguity"] = "NO_CGROUP_V2_MEMBERSHIP"
        adequate = False
    else:
        root = Path("/sys/fs/cgroup")
        current = (root / group.lstrip("/")).resolve()
        assert current.is_relative_to(root)
        for _ in range(16):
            row = {"path": str(current), **{name: small(current / name) for name in (
                "memory.max", "memory.high", "memory.current", "cpu.max", "pids.max", "pids.current",
                "cpuset.cpus.effective")}}
            usage = row["memory.current"]
            if row["memory.max"] is None and current == root:
                row["root_limit_note"] = "NO_ROOT_MEMORY_MAX_FILE; PHYSICAL_HOST_MEMAVAILABLE_GATE_RETAINED"
            elif row["memory.max"] is None or usage is None:
                adequate = False
                row["capacity_ambiguity"] = "MEMORY_CONTROLLER_LIMIT_OR_USAGE_NOT_EXPOSED"
            else:
                for key in ("memory.max", "memory.high"):
                    value = row[key]
                    if value is not None and value != "max":
                        adequate = adequate and int(value) - int(usage) >= 4 * 1024 ** 3
            result["cgroup_ancestors"].append(row)
            if current == root:
                break
            current = current.parent
        else:
            raise AssertionError("CGROUP_ANCESTOR_COUNT_BOUND")
    result["single_constructor_headroom_snapshot_sufficient"] = adequate
    return result


def read_source(path, maximum=262144):
    pin()
    with Path(path).open("rb") as stream:
        raw = stream.read(maximum + 1)
    assert len(raw) <= maximum
    return raw, ast.parse(raw.decode("utf-8"), filename=str(path))


def readonly_helper_preflight(environment, observation, lineage):
    assert lineage["v3_actual_sha256"] == "0efbfc8ec9a4608171d21b89d6c824b13218ded1cb29971dcff5c6836d8a1af5"
    policy = lineage["known_readonly_helper_policy"]
    binary = Path(shutil.which("uname", path=environment.get("PATH"))).resolve()
    assert binary == Path(policy["path"]) == Path("/usr/bin/uname")
    info = os.lstat(binary)
    assert stat.S_ISREG(info.st_mode) and info.st_uid == 0 and not info.st_mode & 0o6022
    assert [info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns] == policy["identity"]
    with binary.open("rb") as stream:
        assert hashlib.sha256(stream.read(1048577)).hexdigest() == policy["sha256"]
    platform = Path(policy["platform_path"])
    assert hash_file(platform) == policy["platform_sha256"]
    pinfo = platform.stat()
    null = os.lstat("/dev/null")
    assert stat.S_ISCHR(null.st_mode) and null.st_rdev == os.makedev(1, 3)
    proof = {"policy": policy, "platform_and_config_source_reused_from_v3_actual": lineage["v3_actual_sha256"],
             "installed_source_recollection": False, "config_route_unchanged": True}
    guards = [{"path": str(binary), "resolved": str(binary), "identity": policy["identity"]},
              {"path": str(platform), "resolved": str(platform.resolve()),
               "identity": [pinfo.st_dev, pinfo.st_ino, pinfo.st_size, pinfo.st_mtime_ns]}]
    return policy, proof, guards


def private_tmp_snapshot(policy, may_remove_empty):
    root = Path(policy["path"])
    info = os.lstat(root)
    assert root.resolve() == root and root.parent == Path("/tmp") and root.name.startswith("m234_constructor_4ecb5020_")
    assert stat.S_ISDIR(info.st_mode) and info.st_uid == 1001 and stat.S_IMODE(info.st_mode) == 0o700
    assert [info.st_dev, info.st_ino] == policy["identity"]
    entries = []
    total = 0
    with os.scandir(root) as stream:
        for entry in stream:
            total += 1
            if len(entries) < 16:
                item = entry.stat(follow_symlinks=False)
                entries.append({"name": entry.name[:160], "directory": stat.S_ISDIR(item.st_mode),
                                "symlink": stat.S_ISLNK(item.st_mode), "bytes": item.st_size})
            if total >= 256:
                break
    record = {"policy": policy, "entries_first16": entries, "entries_count_lower_bound": total,
              "listing_complete": total < 256, "recursive_cleanup": False, "removed_empty": False}
    if total == 0 and may_remove_empty:
        again = os.lstat(root)
        assert [again.st_dev, again.st_ino] == policy["identity"] and stat.S_ISDIR(again.st_mode)
        os.rmdir(root)
        record["removed_empty"] = True
    else:
        record["own_residue_preserved"] = True
    return record


def ipv6_capability_source_preflight(observation):
    path = Path("/home/songcl/.conda/envs/project_cluster/lib/python3.10/site-packages/urllib3/util/connection.py")
    raw, syntax = read_source(path)
    nodes = [node for node in syntax.body if isinstance(node, ast.FunctionDef) and node.name == "_has_ipv6"]
    assert len(nodes) == 1
    node = nodes[0]
    lines = raw.decode("utf-8").splitlines()
    observation["urllib3_ipv6_source_before_validation"] = {"path": str(path), "sha256": hashlib.sha256(raw).hexdigest(),
        "start": node.lineno, "end": node.end_lineno, "complete_source": "\n".join(lines[node.lineno-1:node.end_lineno])}
    assert len(node.args.args) == 1 and node.end_lineno - node.lineno < 80
    parameter = node.args.args[0].arg
    binds = [call for call in ast.walk(node) if isinstance(call, ast.Call) and isinstance(call.func, ast.Attribute)
             and call.func.attr == "bind"]
    assert len(binds) == 1 and len(binds[0].args) == 1
    address = binds[0].args[0]
    assert isinstance(address, ast.Tuple) and len(address.elts) == 2
    assert isinstance(address.elts[0], ast.Name) and address.elts[0].id == parameter
    assert ast.literal_eval(address.elts[1]) == 0
    sockets = [call for call in ast.walk(node) if isinstance(call, ast.Call) and isinstance(call.func, ast.Attribute)
               and isinstance(call.func.value, ast.Name) and call.func.value.id == "socket" and call.func.attr == "socket"]
    assert len(sockets) == 1 and len(sockets[0].args) >= 1
    family = sockets[0].args[0]
    assert isinstance(family, ast.Attribute) and family.attr == "AF_INET6"
    for call in ast.walk(node):
        if isinstance(call, ast.Call) and isinstance(call.func, ast.Attribute):
            assert call.func.attr not in ("connect", "connect_ex", "listen", "send", "sendall", "sendto", "sendmsg", "getaddrinfo")
    calls = [call for statement in syntax.body if not isinstance(statement, (ast.FunctionDef, ast.ClassDef))
             for call in ast.walk(statement) if isinstance(call, ast.Call) and isinstance(call.func, ast.Name)
             and call.func.id == "_has_ipv6"]
    assert len(calls) == 1 and len(calls[0].args) == 1 and ast.literal_eval(calls[0].args[0]) == "::1"
    info = path.stat()
    policy = {"path": str(path), "sha256": hashlib.sha256(raw).hexdigest(),
              "identity": [info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns],
              "maximum_binds": 1, "address": ["::1", 0], "family": "AF_INET6", "type": "SOCK_STREAM"}
    guard = {"path": str(path), "resolved": str(path.resolve()), "identity": policy["identity"]}
    return policy, guard
