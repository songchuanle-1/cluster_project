def sha_file(path):
    before = os.stat(path)
    value = hashlib.sha256()
    with open(path,'rb') as stream:
        for chunk in iter(lambda:stream.read(1048576),b''): value.update(chunk)
    after = os.stat(path)
    assert (before.st_dev,before.st_ino,before.st_size,before.st_mtime_ns) == (after.st_dev,after.st_ino,after.st_size,after.st_mtime_ns)
    return {'path':str(path),'realpath':str(Path(path).resolve()),'sha256':value.hexdigest(),'bytes':before.st_size,
            'device':before.st_dev,'inode':before.st_ino,'uid':before.st_uid,'mtime_ns':before.st_mtime_ns,'read_stable':True}

def run(argv,env=None):
    child = subprocess.Popen(argv,stdout=subprocess.PIPE,stderr=subprocess.PIPE,env=env)
    try: stdout,stderr = child.communicate(timeout=25)
    except subprocess.TimeoutExpired: raise RuntimeError('Readonly child uncertain; preserve PID '+str(child.pid)+'; no signal/retry')
    assert child.returncode == 0,stderr.decode('utf-8','replace')[-1000:]
    return stdout.decode('utf-8','replace')
