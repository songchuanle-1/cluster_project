def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def file_bytes(path,maximum=2097152):
    before = os.lstat(path)
    assert stat.S_ISREG(before.st_mode) and before.st_size <= maximum
    raw = path.read_bytes()
    after = os.lstat(path)
    stable = (before.st_dev,before.st_ino,before.st_size,before.st_mtime_ns,before.st_ctime_ns) == (after.st_dev,after.st_ino,after.st_size,after.st_mtime_ns,after.st_ctime_ns)
    return raw,{'path':str(path),'sha256':sha(raw),'size':before.st_size,'mtime_ns':before.st_mtime_ns,
                'ctime_ns':before.st_ctime_ns,'device':before.st_dev,'inode':before.st_ino,
                'uid':before.st_uid,'read_stable':stable}

def numbered(text,start,end):
    return [{'line':i,'text':x} for i,x in enumerate(text.splitlines(),1) if start <= i <= end]

def functions(code,name):
    found = []
    if code.co_name == name: found.append(code)
    for constant in code.co_consts:
        if isinstance(constant,types.CodeType): found += functions(constant,name)
    return found

def instructions(code):
    current_line = code.co_firstlineno
    result = []
    for instruction in dis.get_instructions(code):
        if instruction.starts_line is not None: current_line = instruction.starts_line
        result.append({'offset':instruction.offset,'opname':instruction.opname,'argrepr':instruction.argrepr,
                       'starts_line':instruction.starts_line,'effective_line':current_line})
    return result
