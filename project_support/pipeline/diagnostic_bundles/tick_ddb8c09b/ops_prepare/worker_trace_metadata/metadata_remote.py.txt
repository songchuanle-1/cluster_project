def identity(pid):
    try:
        return (Path("/proc") / str(pid) / "stat").read_text().rsplit(")", 1)[1].split()[19]
    except FileNotFoundError:
        return None


def fingerprint(info):
    return {"dev": info.st_dev, "ino": info.st_ino, "bytes": info.st_size,
            "mtime_ns": info.st_mtime_ns, "uid": info.st_uid, "mode": stat.S_IMODE(info.st_mode)}


def metadata(path, hash_max=0):
    global HASH_READ
    pin()
    path = Path(path)
    result = {"path": str(path), "resolved": str(path.resolve())}
    try:
        link = os.lstat(path)
        info = os.stat(path)
        result.update(lstat=fingerprint(link), stat=fingerprint(info),
                      regular=stat.S_ISREG(info.st_mode), directory=stat.S_ISDIR(info.st_mode))
        if hash_max and stat.S_ISREG(info.st_mode):
            if info.st_size <= hash_max and HASH_READ + info.st_size <= HASH_BUDGET:
                digest = hashlib.sha256()
                with path.open("rb") as stream:
                    for block in iter(lambda: stream.read(65536), b""):
                        pin()
                        HASH_READ += len(block)
                        assert HASH_READ <= HASH_BUDGET
                        digest.update(block)
                assert fingerprint(info) == fingerprint(os.stat(path)), "FILE_CHANGED_DURING_HASH"
                result["sha256"] = digest.hexdigest()
            else:
                result["hash_omitted"] = "PER_FILE_OR_TOTAL_HASH_BYTE_BOUND"
        result["outcome"] = "OBSERVED"
    except (FileNotFoundError, PermissionError) as error:
        result.update(outcome=type(error).__name__, error=str(error)[:240])
    return result


def require_hash(path, expected):
    item = metadata(path, 32 * 1024 * 1024)
    assert item.get("sha256") == expected, "PINNED_FILE_MISMATCH: " + str(path)
    return item


def text(path, maximum):
    global SOURCE_READ
    pin()
    path = Path(path)
    with path.open("rb") as stream:
        raw = stream.read(maximum + 1)
    assert len(raw) <= maximum, "TEXT_FILE_BYTE_BOUND"
    SOURCE_READ += len(raw)
    assert SOURCE_READ <= 2 * 1024 * 1024, "TOTAL_SOURCE_TEXT_BYTE_BOUND"
    return raw.decode("utf-8", "replace")


def optional_text(path, maximum):
    try:
        value = text(path, maximum)
        return {"path": str(path), "text": value, "bytes": len(value.encode())}
    except (FileNotFoundError, PermissionError) as error:
        return {"path": str(path), "outcome": type(error).__name__}


def source(path, requested):
    global SOURCE_LINES
    item = metadata(path, 512 * 1024)
    if item["outcome"] != "OBSERVED":
        return item
    value = text(path, 512 * 1024)
    rows, tree = value.splitlines(), ast.parse(value, filename=str(path))
    sections = []
    for parent_name, function_name, limit in requested:
        parents = [tree] if parent_name is None else [
            node for node in tree.body if isinstance(node, ast.ClassDef) and node.name == parent_name]
        nodes = [node for parent in parents for node in parent.body
                 if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name == function_name]
        if len(nodes) != 1:
            sections.append({"class": parent_name, "function": function_name, "outcome": "MISSING_OR_AMBIGUOUS"})
            continue
        node = nodes[0]
        selected = rows[node.lineno - 1:min(node.end_lineno, node.lineno - 1 + limit)]
        SOURCE_LINES += len(selected)
        assert SOURCE_LINES <= 1250, "TOTAL_SOURCE_EXCERPT_LINE_BOUND"
        sections.append({"class": parent_name, "function": function_name, "start": node.lineno,
                         "end": node.end_lineno, "complete": node.end_lineno - node.lineno + 1 <= limit,
                         "lines": [{"line": node.lineno + i, "text": line[:320],
                                    "truncated": len(line) > 320} for i, line in enumerate(selected)]})
    item["sections"] = sections
    return item


def manifest(directory, suffix, limit=5000):
    pin()
    directory = Path(directory)
    result = metadata(directory)
    if result["outcome"] != "OBSERVED":
        return result, []
    names, count = [], 0
    with os.scandir(directory) as entries:
        for entry in entries:
            count += 1
            assert count <= limit, "DIRECTORY_ENTRY_COUNT_BOUND: " + str(directory)
            if entry.name.endswith(suffix):
                names.append(entry.name)
    rows = []
    for name in sorted(names):
        pin()
        info = os.stat(directory / name)
        assert stat.S_ISREG(info.st_mode)
        rows.append({"name": name, **fingerprint(info)})
    raw = json.dumps(rows, sort_keys=True, separators=(",", ":")).encode()
    result.update(manifest_sha256=hashlib.sha256(raw).hexdigest(), matching_files=len(rows),
                  directory_entries=count, total_file_bytes=sum(row["bytes"] for row in rows),
                  first_12=rows[:12], complete_metadata_manifest=True, file_contents_read=False)
    return result, rows


def observe(argv, env, seconds, label):
    pin()
    out = os.memfd_create("worker_metadata_stdout", os.MFD_CLOEXEC)
    err = os.memfd_create("worker_metadata_stderr", os.MFD_CLOEXEC)
    process = subprocess.Popen(argv, cwd=str(ROOT), env=env, stdin=subprocess.DEVNULL, stdout=out, stderr=err)
    PROCESSES.append(process)
    start = identity(process.pid)
    until = min(STOP, time.monotonic() + seconds)
    while process.poll() is None and time.monotonic() < until:
        time.sleep(0.02)
    code = process.poll()
    def tail(fd):
        size = os.fstat(fd).st_size
        return os.pread(fd, min(size, 1024), max(0, size - 1024)).decode("utf-8", "replace")
    result = {"label": label, "argv": argv, "pid": process.pid, "start_ticks": start,
              "certainty": "terminal" if code is not None else "uncertain", "returncode": code,
              "stdout_bytes": os.fstat(out).st_size, "stdout": tail(out), "stderr_tail": tail(err),
              "signals_sent": False, "automatic_retry": False}
    if code is None:
        result.update(reason="SOFT_TIMEOUT_PRESERVE_PID_NO_SIGNAL_OR_RETRY",
                      output_descriptors=[f"/proc/{process.pid}/fd/1", f"/proc/{process.pid}/fd/2"])
    else:
        os.close(out)
        os.close(err)
    return result
