def identity():
    return Path("/proc/self/stat").read_text().rsplit(")", 1)[1].split()[19]


def event(stage, **values):
    print(json.dumps({"stage": stage, "pid": os.getpid(), **values}), flush=True)


def check():
    assert time.monotonic() < STOP, "CARRIER_SOFT_BUDGET_BEFORE_NEXT_PHASE"
    assert os.environ.get("CUDA_VISIBLE_DEVICES") == ""
    assert os.environ.get("TOKENIZERS_PARALLELISM") == "false"


def same_file(record):
    path = Path(record["path"])
    info = os.stat(path)
    assert str(path.resolve()) == record["resolved"]
    assert [info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns] == record["identity"]


def guard_snapshot():
    changed = []
    for record in GUARDED_FILES:
        try:
            same_file(record)
        except BaseException as error:
            changed.append({"path": record["path"], "error": str(error)[:160]})
    return {"checked": len(GUARDED_FILES), "unchanged": not changed, "changed_first6": changed[:6]}


def forbidden(name):
    def stop(*args, **kwargs):
        event("forbidden_operation", status="BLOCKED_FORBIDDEN_OPERATION", phase=PHASE, operation=name,
              guarded_file_identity_after=guard_snapshot())
        faulthandler.disable()
        sys.stdout.flush()
        sys.stderr.flush()
        os._exit(4)
    return stop


def verified_uname():
    policy = HELPER_POLICY
    path = Path(policy["path"])
    assert path.resolve() == path and str(path) == "/usr/bin/uname"
    info = os.lstat(path)
    assert stat.S_ISREG(info.st_mode) and info.st_uid == 0 and not info.st_mode & 0o6022
    assert [info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns] == policy["identity"]
    assert shutil.which("uname") is not None and Path(shutil.which("uname")).resolve() == path
    with path.open("rb") as stream:
        digest = hashlib.sha256(stream.read(1024 * 1024 + 1)).hexdigest()
    assert info.st_size <= 1024 * 1024 and digest == policy["sha256"]
    return str(path)


class ObservedReadonlyPopen(OriginalPopen):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)  # Audit permits only authenticated platform uname -p.
        self.readonly_identity = Path("/proc") / str(self.pid) / "stat"
        try:
            ticks = self.readonly_identity.read_text().rsplit(")", 1)[1].split()[19]
        except FileNotFoundError:
            ticks = None
        HELPER_RECORDS.append({"pid": self.pid, "start_ticks": ticks, "argv": ["uname", "-p"]})
        event("readonly_helper_started", helper=HELPER_RECORDS[-1])

    def communicate(self, input=None, timeout=None):
        try:
            value = super().communicate(input=input, timeout=5 if timeout is None else min(timeout, 5))
        except subprocess.TimeoutExpired:
            event("readonly_helper_uncertain", status="UNCERTAIN_READONLY_HELPER_PRESERVE_PID_NO_SIGNAL_OR_RETRY",
                  helper=HELPER_RECORDS[-1])
            faulthandler.disable()
            sys.stdout.flush(); sys.stderr.flush()
            os._exit(5)  # Bypass subprocess.run timeout cleanup; do not signal the unknown helper.
        HELPER_RECORDS[-1]["returncode"] = self.returncode
        event("readonly_helper_terminal", helper=HELPER_RECORDS[-1])
        return value


def private_tmp_valid():
    try:
        root = Path(PRIVATE_TMP["path"])
        info = os.lstat(root)
        return (root.is_absolute() and root.resolve() == root and root.parent == Path("/tmp")
                and root.name.startswith("m234_constructor_51fe2245_") and stat.S_ISDIR(info.st_mode)
                and info.st_uid == 1001 and stat.S_IMODE(info.st_mode) == 0o700
                and [info.st_dev, info.st_ino] == PRIVATE_TMP["identity"])
    except (OSError, KeyError):
        return False


def inside_private(path, directory_fd=None):
    try:
        if not private_tmp_valid():
            return False
        root = Path(PRIVATE_TMP["path"])
        if isinstance(path, int):
            raw = os.readlink("/proc/self/fd/" + str(path))
            path = raw[:-10] if raw.endswith(" (deleted)") else raw
        else:
            path = os.fsdecode(path)
        candidate = Path(path)
        if not candidate.is_absolute():
            if isinstance(directory_fd, int) and directory_fd >= 0:
                base = os.readlink("/proc/self/fd/" + str(directory_fd))
                candidate = Path(base) / candidate
            else:
                candidate = Path.cwd() / candidate
        candidate = candidate.resolve(strict=False)
        return candidate != root and candidate.is_relative_to(root)
    except (OSError, TypeError, ValueError):
        return False


def private_operation(event_name, arguments):
    if event_name == "open":
        return inside_private(arguments[0])
    if event_name in ("os.remove", "os.rmdir"):
        return inside_private(arguments[0], arguments[1])
    if event_name in ("os.mkdir", "os.chmod"):
        return inside_private(arguments[0], arguments[2])
    if event_name == "os.utime":
        return inside_private(arguments[0], arguments[3])
    if event_name == "os.rename":
        return inside_private(arguments[0], arguments[2]) and inside_private(arguments[1], arguments[3])
    return False


def audit(event_name, arguments):
    global HELPER_COUNT, TEMP_OP_COUNT
    blocked = event_name in {
        "os.mkdir", "os.rmdir", "os.remove", "os.rename", "os.link", "os.symlink", "os.chmod",
        "os.chown", "os.utime", "os.truncate", "os.chdir", "os.system", "subprocess.Popen",
        "os.fork", "os.forkpty", "os.posix_spawn", "os.posix_spawnp", "os.kill", "os.killpg",
        "socket.connect", "socket.bind", "socket.sendto", "socket.sendmsg", "socket.getaddrinfo",
        "pickle.find_class",
    }
    if event_name == "open":
        mode, flags = arguments[1], arguments[2]
        blocked = (isinstance(mode, str) and any(value in mode for value in ("w", "a", "x", "+")))
        if isinstance(flags, int):
            blocked = blocked or bool(flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND))
    if blocked and private_operation(event_name, arguments):
        blocked = False
        TEMP_OP_COUNT += 1
        if TEMP_OP_COUNT <= 12:
            event("allowed_owned_temporary_operation", phase=PHASE, audit_event=event_name,
                  target=str(arguments[0])[:240], owned_private_tmp=PRIVATE_TMP["path"])
    if event_name == "open" and blocked and isinstance(arguments[0], (str, bytes)):
        if os.fsdecode(arguments[0]) == "/dev/null":
            info = os.lstat("/dev/null")
            flags = arguments[2]
            if (stat.S_ISCHR(info.st_mode) and info.st_rdev == os.makedev(1, 3)
                    and isinstance(flags, int) and not flags & (os.O_CREAT | os.O_TRUNC | os.O_APPEND)):
                blocked = False
                event("allowed_null_device_open", phase=PHASE, rdev_major=1, rdev_minor=3)
    if event_name == "subprocess.Popen":
        executable, argv, cwd, environment = arguments
        stack = traceback.extract_stack(limit=14)
        authentic_caller = any(row.filename == HELPER_POLICY["platform_path"] and row.name == "from_subprocess"
                               for row in stack)
        if (executable == "uname" and isinstance(argv, (list, tuple)) and list(argv) == ["uname", "-p"]
                and cwd is None and environment is None and authentic_caller and HELPER_COUNT == 0):
            resolved = verified_uname()
            HELPER_COUNT += 1
            blocked = False
            event("allowed_readonly_helper", argv=["uname", "-p"], resolved=resolved,
                  sha256=HELPER_POLICY["sha256"], platform_source_sha256=HELPER_POLICY["platform_sha256"])
    if event_name.startswith(("os.spawn", "os.exec")):
        blocked = True
    if blocked:
        target = str(arguments[0])[:240] if arguments else None
        event("audit_blocked", status="BLOCKED_AUDIT_EVENT", phase=PHASE, audit_event=event_name,
              target=target, guarded_file_identity_after=guard_snapshot(),
              stack=[{"file": row.filename, "line": row.lineno, "function": row.name}
                                   for row in traceback.extract_stack(limit=7)])
        faulthandler.disable()
        sys.stdout.flush()
        sys.stderr.flush()
        os._exit(3)  # Own carrier only; no library fallback or finalizer after a denied operation.

        assert os.readlink("/proc/self/fd/" + str(descriptor)).startswith("/memfd:constructor_v1_" + kind)

    os.dup2(out, 1)
    os.dup2(err, 2)
    os.close(out)
    os.close(err)

    STOP = min(payload["stop_monotonic"], time.monotonic() + 60)

    HELPER_POLICY = payload["helper_policy"]
    PRIVATE_TMP = payload["private_tmp"]

    code = 2
    try:

        assert site.ENABLE_USER_SITE is True and Path.cwd() == ROOT

            same_file(record)

        verified_uname()
        subprocess.Popen = ObservedReadonlyPopen
        sys.addaudithook(audit)  # Active before YAML, Torch or project/library imports.
        event("audit_guard_installed", persistent_writes_network_arbitrary_subprocess_and_unpickle_class_lookup_denied=True,
              exact_platform_uname_p_exception=True)
        sys.path.insert(0, str(ROOT))
        sys.path.insert(0, str(ROOT / "framework/lib"))
        PHASE = "torch_import"

              cuda_initialized=False)

        assert cfg.train.seed == 1999
        train = cfg.data.train

        assert train.data_root.rstrip("/") == str(DATA) and train.prefix == "train"
        assert train.training is True and train.aug is True and train.with_elastic is True and train.mode == 4
        assert cfg.dataloader.train.batch_size == 2 and cfg.dataloader.train.num_workers == 4
        assert cfg.dataloader.train.persistent_workers is True
        check()
        PHASE = "dataset_import"
        event("phase_enter", phase=PHASE)
        from framework.dataset import build_dataset, ScanNetDataset_sample_graph_edge
        for name in ("load", "_torch_load_compat", "__getitem__", "collate_fn", "transform_train",
                     "transform_test", "data_aug", "elastic", "get_sp_label", "get_ref_mask",

              guarded_file_identity_after=guard_snapshot())
        traceback.print_exc(limit=10)
    finally:
        faulthandler.disable()
        sys.stdout.flush()
        sys.stderr.flush()
        os._exit(code)  # Own process; no worker/child exists and no library finalizer is needed.
