def pin():
    assert socket.gethostname() == "dl04-MS-7E06"
    assert os.getuid() == os.geteuid() == 1001
    assert datetime.now(timezone.utc) < datetime.fromisoformat(OWNER["deadline_at"])


def identity(pid):
    try:
        return (Path("/proc") / str(pid) / "stat").read_text().rsplit(")", 1)[1].split()[19]
    except FileNotFoundError:
        return None


def hash_file(path):
    assert path.resolve() == path
    before = os.lstat(path)
    assert stat.S_ISREG(before.st_mode) and before.st_uid == 1001 and before.st_size <= 32 * 1024 * 1024
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(65536), b""):
            digest.update(block)
    after = os.lstat(path)
    assert (before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns) == (
        after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns)
    return digest.hexdigest()


def tail(fd, maximum):
    size = os.fstat(fd).st_size
    return os.pread(fd, min(size, maximum), max(0, size - maximum)).decode("utf-8", "replace")


def observe(argv, environment, seconds, stop, label):
    pin()
    assert time.monotonic() < stop
    out = os.memfd_create("rve_resource_stdout", os.MFD_CLOEXEC)
    err = os.memfd_create("rve_resource_stderr", os.MFD_CLOEXEC)
    process = subprocess.Popen(argv, cwd=str(ROOT), env=environment, stdin=subprocess.DEVNULL,
                               stdout=out, stderr=err)
    PROCESSES.append(process)
    start = identity(process.pid)
    until = min(stop, time.monotonic() + seconds)
    while process.poll() is None and time.monotonic() < until:
        time.sleep(0.02)
    code = process.poll()
    record = {"label": label, "argv": argv, "pid": process.pid, "start_ticks": start,
              "certainty": "terminal" if code is not None else "uncertain", "returncode": code,
              "stdout_bytes": os.fstat(out).st_size, "stderr_bytes": os.fstat(err).st_size,
              "stdout": tail(out, 1024), "stderr_tail": tail(err, 1024),
              "signals_sent": False, "automatic_retry": False}
    if code is None:
        record["reason"] = "SOFT_TIMEOUT_PRESERVE_PID_NO_SIGNAL_OR_RETRY"
        record["output_descriptors"] = [f"/proc/{process.pid}/fd/1", f"/proc/{process.pid}/fd/2"]
    else:
        os.close(out)
        os.close(err)
    return record

        result["locks"] = []
        for kind, inode in (("training", 92537152), ("diagnostic", 92537153), ("exclusive", 92537154)):

            before = os.lstat(path)
            assert stat.S_ISREG(before.st_mode) and before.st_uid == 1001

            fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)

            info, after = os.fstat(fd), os.lstat(path)
            assert (info.st_dev, info.st_ino) == (after.st_dev, after.st_ino) == (66308, inode)

                                    "nonblocking_acquired": available, "created_or_replaced": False})

        if not all_locks:
            for fd in owned:

            owned.clear()
        result["all_original_locks_held_during_query"] = all_locks
