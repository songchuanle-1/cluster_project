def pin():
    assert socket.gethostname() == "dl04-MS-7E06" and os.getuid() == os.geteuid() == 1001

def hash_file(path):
    assert path.resolve() == path
    info = os.lstat(path)
    assert stat.S_ISREG(info.st_mode) and info.st_uid == 1001 and info.st_size <= 32 * 1024 * 1024
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for value in iter(lambda: stream.read(65536), b""):
            digest.update(value)
    after = os.lstat(path)
    assert (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns) == (
        after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns)
    return digest.hexdigest()


def identity(pid):
    try:
        return (Path("/proc") / str(pid) / "stat").read_text().rsplit(")", 1)[1].split()[19]
    except FileNotFoundError:
        return None


def tail(fd, maximum):
    size = os.fstat(fd).st_size
    return os.pread(fd, min(size, maximum), max(0, size - maximum)).decode("utf-8", "replace")


def observe(argv, environment, payload, seconds, global_stop, label):
    pin()
    out = os.memfd_create("rve_native_stdout", os.MFD_CLOEXEC)
    err = os.memfd_create("rve_native_stderr", os.MFD_CLOEXEC)
    process = subprocess.Popen(argv, cwd=str(ROOT), env=environment,
                               stdin=subprocess.PIPE, stdout=out, stderr=err)
    PROCESSES.append(process)
    start = identity(process.pid)
    if payload is not None:
        encoded = json.dumps(payload).encode()
        assert len(encoded) <= 16384
        process.stdin.write(encoded)
    process.stdin.close()
    until = min(global_stop, time.monotonic() + seconds)
    while process.poll() is None and time.monotonic() < until:
        time.sleep(0.02)
    returncode = process.poll()
    stdout = tail(out, 65536)
    records = []
    for line in stdout.splitlines():
        try:
            records.append(json.loads(line))
        except ValueError:
            pass
    result = {"label": label, "pid": process.pid, "start_ticks": start,
              "certainty": "terminal" if returncode is not None else "uncertain",
              "returncode": returncode, "stdout_bytes": os.fstat(out).st_size,
              "stderr_bytes": os.fstat(err).st_size,
              "records": records[-32:], "faulthandler_stderr_tail": tail(err, 3072),
              "stderr_truncated": os.fstat(err).st_size > 3072, "signals_sent": False}
    if returncode is None:
        result["reason"] = "SOFT_TIMEOUT_PRESERVE_PID_NO_RETRY"
        result["output_FDs_if_process_remains"] = [f"/proc/{process.pid}/fd/1", f"/proc/{process.pid}/fd/2"]
    else:
        os.close(out)
        os.close(err)
    return result, stdout
