def failure(error):
    return {"available": False, "error_type": type(error).__name__, "error": str(error)[:600]}

def command(label, executable, arguments, cap=32768, timeout=8):
    pin()
    assert 0 < cap <= 32768 and 0 < timeout <= 8
    program = shutil.which(executable, path="/usr/sbin:/usr/bin:/sbin:/bin")
    record = {"label": label, "argv": [executable, *arguments], "capture_cap_bytes": cap}
    if not program:
        return {**record, "available": False, "certainty": "terminal", "reason": "EXECUTABLE_NOT_FOUND"}
    env = {"PATH": "/usr/sbin:/usr/bin:/sbin:/bin", "LC_ALL": "C", "LANG": "C", "TZ": "UTC",
           "SYSTEMD_PAGER": "cat", "SYSTEMD_COLORS": "0", "PAGER": "cat",
           "PYTHONDONTWRITEBYTECODE": "1"}
    try:
        process = subprocess.Popen([program, *arguments], stdin=subprocess.DEVNULL,
                                   stdout=subprocess.PIPE, stderr=subprocess.PIPE, env=env)
    except OSError as error:
        return {**record, **failure(error), "certainty": "terminal"}
    PROCESSES.append(process)  # Keep uncertain children/pipe handles; never terminate or kill.
    record.update(available=True, pid=process.pid, certainty="uncertain")
    selector = selectors.DefaultSelector()
    buffers = {"stdout": bytearray(), "stderr": bytearray()}
    for name, stream in (("stdout", process.stdout), ("stderr", process.stderr)):
        os.set_blocking(stream.fileno(), False)
        selector.register(stream, selectors.EVENT_READ, name)
    deadline = time.monotonic() + timeout
    truncated = False
    try:
        while selector.get_map() and time.monotonic() < deadline:
            for key, _ in selector.select(timeout=min(0.1, max(0, deadline - time.monotonic()))):
                try:
                    raw = os.read(key.fileobj.fileno(), 4096)
                except BlockingIOError:
                    continue
                if not raw:
                    selector.unregister(key.fileobj)
                    continue
                room = max(0, cap - sum(map(len, buffers.values())))
                buffers[key.data].extend(raw[:room])
                truncated |= len(raw) > room
        returncode = process.poll()
        if returncode is not None and not selector.get_map():
            record.update(certainty="terminal", returncode=returncode)
        else:
            record["reason"] = "TIMEOUT_OR_PENDING_PIPE; PID retained, no signal/retry"
    finally:
        selector.close()  # Does not close child pipes or signal the child.
    record.update({name: bytes(raw).decode("utf-8", "replace") for name, raw in buffers.items()})
    record.update(captured_bytes=sum(map(len, buffers.values())), output_truncated=truncated,
                  signals_sent=False)
    return record

def report_header(path):
    fd = None
    try:
        pin()
        fd = os.open(path, os.O_RDONLY | os.O_CLOEXEC | os.O_NOFOLLOW)
        info = os.fstat(fd)
        assert stat.S_ISREG(info.st_mode) and info.st_uid == 1001
        result = {"path": str(path), "bytes": info.st_size, "mtime_ns": info.st_mtime_ns,
                  "uid": info.st_uid, "report_header_budget_bytes": 65536}
        fields, current, consumed, stopped_at_core = {}, None, 0, False
        allowed = {"ProblemType", "Date", "ExecutablePath", "ExecutableTimestamp",
                   "Signal", "ProcStatus", "ProcCwd", "ProcCmdline", "Uname", "ProcPID"}

        def byte():
            nonlocal consumed
            if consumed >= 65536:
                return b""
            value = os.read(fd, 1)  # Unbuffered: never prefetch encoded CoreDump data.
            consumed += len(value)
            return value

        def rest_of_line(first=b""):
            value = bytearray(first)
            while consumed < 65536:
                item = byte()
                if not item or item == b"\n":
                    break
                if len(value) < 4096:
                    value.extend(item)
            return bytes(value)

        while consumed < 65536:
            item = byte()
            if not item:
                break
            if item == b"\n":
                current = None
                continue
            if item in (b" ", b"\t"):
                continuation = rest_of_line().decode("utf-8", "replace")
                if current in allowed:
                    fields[current] = (fields.get(current, "") + "\n" + continuation)[:8192]
                continue
            key = bytearray(item)
            while consumed < 65536:
                item = byte()
                if not item or item in (b":", b"\n"):
                    break
                assert len(key) < 96, "Malformed report header key"
                key.extend(item)
            if item != b":":
                current = None
                continue
            current = bytes(key).decode("ascii", "strict")
            if current == "CoreDump":
                stopped_at_core = True
                break  # Stop at the key/colon, before reading any value or core bytes.
            value = rest_of_line().decode("utf-8", "replace").lstrip()
            if current in allowed:
                fields[current] = value[:8192]
        status = fields.get("ProcStatus", "")
        pid_values = re.findall(r"(?m)^\s*Pid:\s*(\d+)", status)
        uid_values = re.findall(r"(?m)^\s*Uid:\s*(\d+)", status)
        if fields.get("ProcPID", "").isdigit():
            pid_values.append(fields["ProcPID"])
        pid_match = any(int(value) in TARGET_PIDS for value in pid_values)
        uid_match = bool(uid_values and int(uid_values[0]) == 1001)
        exe_match = fields.get("ExecutablePath") in PYTHONS
        after = os.fstat(fd)
        assert (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns) == (
            after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns)
        result.update(header_bytes_read=consumed, stopped_before_core_value=stopped_at_core,
                      header_budget_exhausted=consumed >= 65536, target_pid_match=pid_match,
                      report_uid_match=uid_match, exact_python_executable_match=exe_match,
                      core_contents_read_or_decoded=False)
        if pid_match and uid_match and exe_match:
            selected_status = [line.strip() for line in status.splitlines() if re.match(
                r"\s*(Name|State|Tgid|Pid|PPid|TracerPid|Uid|Gid|Threads|CoreDumping):", line)]
            result["matching_report_fields"] = {
                key: value[:1024] for key, value in fields.items() if key != "ProcStatus"}
            result["matching_report_status_lines"] = selected_status[:16]
        else:
            result["reason"] = "HEADER_NOT_PROVEN_TO_BE_EXACT_TARGET; NO_FIELD_VALUES_EMITTED"
        return result
    except FileNotFoundError:
        return {"path": str(path), "exists": False}
    except Exception as error:
        return {"path": str(path), **failure(error), "core_contents_read_or_decoded": False}
    finally:
        if fd is not None:
            os.close(fd)

def apport_log():
    path = Path("/var/log/apport.log")
    fd = None
    try:
        pin()
        fd = os.open(path, os.O_RDONLY | os.O_CLOEXEC | os.O_NOFOLLOW)
        info = os.fstat(fd)
        assert stat.S_ISREG(info.st_mode)
        offset = max(0, info.st_size - 32768)
        os.lseek(fd, offset, os.SEEK_SET)
        raw = os.read(fd, 32768)
        matches = [line[:800] for line in raw.decode("utf-8", "replace").splitlines()
                   if PID_PATTERN.search(line)]
        return {"path": str(path), "available": True, "bytes": info.st_size,
                "tail_offset": offset, "bytes_read": len(raw),
                "only_exact_target_PID_lines": matches[-16:],
                "unmatched_lines_emitted": False, "complete_historical_log_scanned": False}
    except Exception as error:
        return {"path": str(path), **failure(error)}
    finally:
        if fd is not None:
            os.close(fd)
